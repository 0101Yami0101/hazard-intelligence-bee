# and needs none of the geospatial or ML stack — no rasterio, geopandas,
# lightgbm, scikit-image or pysheds. That is what lets it run on a free 1 GB
# host, and it stayed true when FloodSense was added: its terrain analysis runs
# offline in scripts/build/ and ships as plain numpy arrays.
#
# folium/streamlit-folium ARE pure Python (no GDAL), so the interactive map
# costs nothing in host compatibility. An earlier build left them out and
# shipped a static PNG with no zoom, pan or click — that was a mistake.

pandas>=2.1
pillow>=10.0
requests>=2.31
# >=1.45 for st.selectbox(accept_new_options=…) — that is what lets the search
# box take a raw latitude/longitude, not just a name from the gazetteer.
streamlit>=1.60      # download_button built on click (the forecast report)
folium>=0.17
streamlit-folium>=0.24   # pixelated=True — nearest-neighbour overlay
                         # rendering, so the ~0.55 km cells keep hard
                         # edges instead of blurring into each other
